  <script>
    // Array que armazena os serviços/peças
    let items = [];

    // Função disparada ao clicar nas "Abinhas"
    function addChipItem(desc, price) {
      items.push({ desc: desc, price: price, qty: 1 });
      renderItemsList();
    }

    // Função disparada no botão "Adicionar Item Manual"
    function addNewItem() {
      items.push({ desc: '', price: '', qty: 1 });
      renderItemsList();
    }

    // Remove a linha selecionada
    function removeItem(index) {
      items.splice(index, 1);
      renderItemsList();
    }

    // Calcula o valor total
    function applyQtyRule() {
      let total = 0;
      items.forEach(item => {
        let p = parseFloat(item.price) || 0;
        let q = parseInt(item.qty) || 1;
        total += (p * q);
      });
      // Formata para o padrão brasileiro (vírgula para centavos)
      document.getElementById('totalValue').textContent = total.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }

    // Renderiza a lista na tela
    function renderItemsList() {
      const container = document.getElementById('itemsContainer');
      container.innerHTML = '';

      items.forEach((item, index) => {
        var row = document.createElement('div');
        row.className = 'item-row';

        // LÓGICA DO PASSO 2: Condição para exibir Texto ou Campo de Digitação
        if (item.desc && item.desc.trim() !== '') {
          // Se tem descrição (veio do botão rápido), mostra apenas o texto
          var dText = document.createElement('div');
          dText.style.fontSize = '14px';
          dText.style.fontWeight = '600';
          dText.style.flex = '1';
          dText.style.color = '#334E68';
          dText.textContent = item.desc;
          row.appendChild(dText);
        } else {
          // Se não tem descrição (veio do item manual), exibe input
          var dInput = document.createElement('input');
          dInput.type = 'text'; 
          dInput.className = 'input-desc';
          dInput.placeholder = 'Descrição...';
          dInput.value = item.desc;
          dInput.addEventListener('input', function(){ 
            item.desc = dInput.value; 
          });
          row.appendChild(dInput);
        }

        // Input Preço
        var pInput = document.createElement('input');
        pInput.type = 'number';
        pInput.className = 'input-price';
        pInput.placeholder = 'R$';
        pInput.value = item.price;
        pInput.addEventListener('input', function(){
          item.price = pInput.value;
          applyQtyRule();
        });
        row.appendChild(pInput);

        // Input Quantidade
        var qInput = document.createElement('input');
        qInput.type = 'number';
        qInput.className = 'input-qty';
        qInput.min = '1';
        qInput.value = item.qty;
        qInput.addEventListener('input', function(){
          item.qty = qInput.value;
          applyQtyRule();
        });
        row.appendChild(qInput);

        // Botão X
        var rmBtn = document.createElement('button');
        rmBtn.className = 'btn-remove';
        rmBtn.textContent = '✕';
        rmBtn.onclick = function() { removeItem(index); };
        row.appendChild(rmBtn);

        container.appendChild(row);
      });

      applyQtyRule();
    }

    // Adiciona uma linha vazia assim que o site carrega
    window.onload = function() {
      addNewItem();
    };
  </script>
